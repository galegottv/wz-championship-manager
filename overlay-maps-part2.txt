}

// ── Ativar mapa ───────────────────────────────────────
function goTo(idx, animate) {
  if (!MAPS.length) return;
  idx = Math.max(0, Math.min(idx, MAPS.length - 1));
  currentIdx = idx;

  if (animate) {
    const fl = document.getElementById('drop-flash');
    fl.classList.add('on');
    setTimeout(() => fl.classList.remove('on'), 200);
  }

  document.querySelectorAll('.slide').forEach((s, i) => {
    s.classList.remove('active','entering');
    if (i === idx) {
      if (animate) {
        s.classList.add('entering');
        setTimeout(() => { s.classList.remove('entering'); s.classList.add('active'); }, 550);
      } else {
        s.classList.add('active');
      }
    }
  });

  document.querySelectorAll('.mdot').forEach((d, i) => {
    d.classList.remove('done','active');
    d.style.width = '16px';
    if (i < idx)       { d.classList.add('done'); }
    else if (i === idx){ d.classList.add('active'); d.style.width = '32px'; }
  });
}

// ── Refresh tourney info ──────────────────────────────
function refreshInfo() {
  const S = getState();
  document.querySelectorAll('[data-info="name"]').forEach(el =>
    el.textContent = (S.name || 'WZ CHAMPIONSHIP').toUpperCase()
  );
  document.querySelectorAll('[data-info="phase"]').forEach(el =>
    el.textContent = (S.phase || 'MAP ROTATION').toUpperCase()
  );
}

// ── Watch localStorage (polling 300ms) ───────────────
let lastDropIdx  = -1;
let lastMode     = '';

function checkState() {
  const S   = getState();
  const mode = S.mode || 'resurgence';
  const idx  = Math.max(0, Math.min(parseInt(localStorage.getItem(LS_DROP)||'0'), (ALL_MAPS[mode]||[]).length - 1));

  // Modo mudou → rebuild completo
  if (mode !== lastMode) {
    lastMode    = mode;
    lastDropIdx = -1;
    buildSlides(S);
    goTo(idx, false);
    lastDropIdx = idx;
    return;
  }

  // Só o índice mudou
  if (idx !== lastDropIdx) {
    const animate = lastDropIdx >= 0;
    lastDropIdx = idx;
    goTo(idx, animate);
  }
}

// ── Init ─────────────────────────────────────────────
(function init() {
  const S    = getState();
  const mode = S.mode || 'resurgence';
  const idx  = parseInt(localStorage.getItem(LS_DROP)||'0');
  lastMode    = mode;
  lastDropIdx = idx;
  buildSlides(S);
  goTo(Math.max(0, Math.min(idx, MAPS.length-1)), false);
  setInterval(checkState,  300);
  setInterval(refreshInfo, 5000);
  window.addEventListener('storage', checkState);
})();
</script>
